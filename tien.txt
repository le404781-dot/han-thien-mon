      interaction_count INTEGER NOT NULL DEFAULT 0,

async function tienDaoRegionAccessFor(db,userId){
  const u=(await db.query(`SELECT id,username FROM users WHERE id=$1`,[userId])).rows[0];
  if(!u)return false;
  const setting=(await db.query(`SELECT region_open FROM tien_dao_settings WHERE singleton_id=1`)).rows[0];
  if(setting?.region_open!==false)return true;
  // Tiên Dao có khóa vùng độc lập. Khi bị khóa, chỉ Đan Chủ được phép vào
  // để quản trị/mở vùng; không dùng quyền bypass của các khu vực khác.
  const owner=(await db.query(`SELECT user_id FROM venue_roles WHERE venue_code='dan-duong' LIMIT 1`)).rows[0]?.user_id||0;
  return Number(owner)===Number(userId);
}

    await ensureTienDaoSchema();

    if(Number(owner)===Number(uid)){await client.query('ROLLBACK');return res.status(400).json({error:'Đan Chủ không cần gửi lời mời.'});}
    const existing=(await client.query(`SELECT id FROM tien_dao_events WHERE user_id=$1 AND npc_code='da_nguyet' AND event_key LIKE 'invite_pending:%' AND created_at>NOW()-INTERVAL '10 minutes' LIMIT 1 FOR UPDATE`,[uid])).rows[0];

    try{await createMailboxNotification(owner,'tien_dao_invite','🌙 Lời mời Tiên Dao','Môn nhân muốn đối thoại với Dạ Nguyệt và Bạch Nguyệt khi quầy đang đóng. Hãy mở Hòm Thư để đồng thuận hoặc từ chối.','#dan-duong');}

  }catch(e){try{await client.query('ROLLBACK')}catch{}console.error('tien-dao invite:',e);res.status(500).json({error:'Không thể gửi lời mời.'});}
  finally{client.release();}
});
